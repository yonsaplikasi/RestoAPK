function processPaymentData(orderId, paymentDetails) { const orderSheet = getSheet("Order"); const paymentSheet = getSheet("Pembayaran"); const data = orderSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === orderId) { // Ubah status order menjadi Selesai orderSheet.getRange(i + 1, 4).setValue("Selesai"); // Catat log transaksi pembayaran paymentSheet.appendRow([ "PAY" + String(new Date().getTime()).slice(-6), orderId, paymentDetails.metode, // CASH, QRIS, atau TRANSFER paymentDetails.total, new Date() ]); return { success: true, message: "Pembayaran berhasil diproses" }; } } return { success: false, message: "Order tidak ditemukan" }; }
